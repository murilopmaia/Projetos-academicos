create database CatCoffe;



create table Reserva(
	Cod_Reserva int auto_increment primary key,
    CPF_ClienteFK varchar(15),
    Cod_GatoFK int,
    Cod_ComboFK int,
    horario_inicio datetime,
    horario_fim datetime,
    Status_Pagamento varchar(20),
    Numero_Mesa int,
    foreign key(CPF_ClienteFK) references Cliente(CPF_Cliente)
    on delete cascade,
    foreign key(Cod_GatoFK) references Gato(Cod_Gato)
    on delete set null,
    foreign key(Cod_ComboFK) references Combo(Cod_Combo)
    on delete set null
);

create table Combo(
	Cod_Combo int auto_increment primary Key,
    Bebida varchar(100),
    Comida varchar(100),
    Disponibilidade boolean,
    Valor float
);


create table Gato(
	Cod_Gato int auto_increment primary key ,
    Nome_Gato varchar(20),
    raca varchar(20),
    idade_Gato int
);

create table Cliente(
	CPF_Cliente varchar(15) primary key,
	Nome_Cliente varchar(50),
    idade_Cliente int,
    telefone_Cliente varchar(15)
);
